🏦 Dart Constructor & Enkapsulasi RekeningBank Demo

class RekeningBank {
  String _nomorRekening;
  double _saldo;

  // Constructor Berparameter
  RekeningBank(this._nomorRekening, this._saldo);

  // Named Constructor: RekeningBank.tanpaSaldo()
  RekeningBank.tanpaSaldo(String nomor)
      : _nomorRekening = nomor,
        _saldo = 0.0;

  // Getter & Setter
  String get nomorRekening => _nomorRekening;
  set nomorRekening(String value) {
    if (value.isNotEmpty) _nomorRekening = value;
  }

  double get saldo => _saldo;

  void deposit(double jml) {
    if (jml > 0) {
      _saldo += jml;
      print('Deposit Rp $jml Berhasil.');
    }
  }

  void tarik(double jml) {
    if (jml > 0 && _saldo >= jml) {
      _saldo -= jml;
      print('Tarik Rp $jml Berhasil.');
    } else {
      print('Gagal: Saldo tidak cukup atau input tidak valid.');
    }
  }

  void cekSaldo() {
    print('Rekening: $_nomorRekening | Saldo: Rp $_saldo');
  }
}

void main() {
  print('--- TRANSAKSI PERTAMA ---');
  var rekeningSaya = RekeningBank('1234-5678-90', 5000000);
  rekeningSaya.cekSaldo();
  
  print('\n--- MELAKUKAN PENARIKAN ---');
  rekeningSaya.tarik(1000000);
  rekeningSaya.cekSaldo();
}
💡 Catatan: Constructor berparameter + named constructor (tanpa saldo) + enkapsulasi dengan getter/saldo.

📌 Kesimpulan


1. Dart memiliki berbagai jenis constructor untuk fleksibilitas objek.

2. Enkapsulasi melindungi data dengan menggunakan private field (prefix underscore).

3. Getter & setter memberikan kontrol akses yang aman terhadap data private
