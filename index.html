⚠️ Masalah Tanpa Extension Methods

// Utility class yang kurang natural
class StringUtils {
    static String capitalize(String text) {
        if (text.isEmpty) return text;
        return text[0].toUpperCase() + text.substring(1);
    }
    static bool isEmail(String text) { return text.contains('@'); }
}
void main(){
    var name = 'john';
    var capitalized = StringUtils.capitalize(name);
    var isEmail = StringUtils.isEmail('test@mail.com');
    print('Original name: $name');
    print('Capitalized name: $capitalized');
    print('Is "test@mail.com" an email? $isEmail');
}
🐾 Penggunaan kurang natural & tidak terintegrasi dengan objek.

✨ Solusi dengan Extension Methods

extension StringExtensions on String {
    String capitalize() {
        if (isEmpty) return this;
        return this[0].toUpperCase() + substring(1);
    }
    bool get isEmail => contains('@');
    String reverse() => split('').reversed.join();
}
void main(){
    var name = 'john';
    var capitalized = name.capitalize(); // 'John'
    var isEmail = 'test@mail.com'.isEmail;
    var reversed = 'hello'.reverse();
    print('Original name: $name');
    print('Capitalized name: $capitalized');
    print('Is "test@mail.com" an email? $isEmail');
    print('Reversed "hello": $reversed');
}

✨ Sintaks Extension Method


extension DateTimeExtensions on DateTime {
  // Method
  String toFormattedString() {
    return '$day/$month/$year';
  }
  
  // Getter
  bool get isToday {
    var now = DateTime.now();
    return year == now.year && 
           month == now.month && 
           day == now.day;
  }
  
  // Operator
  DateTime operator +(Duration duration) {
    return add(duration);
  }
}

void main() {
  // 1. Implementasi toFormattedString()
  print('--- toFormattedString() ---');
  final now = DateTime.now();
  print('Current date: ${now.toFormattedString()}');

  final specificDate = DateTime(2023, 10, 26);
  print('Specific date (2023-10-26): ${specificDate.toFormattedString()}');

  print('\n--- isToday ---');
  // 2. Implementasi isToday
  final today = DateTime.now();
  final tomorrow = DateTime.now().add(Duration(days: 1));
  final yesterday = DateTime.now().subtract(Duration(days: 1));

  print('Is ${today.toFormattedString()} today? ${today.isToday}');
  print('Is ${tomorrow.toFormattedString()} today? ${tomorrow.isToday}');
  print('Is ${yesterday.toFormattedString()} today? ${yesterday.isToday}');

  print('\n--- operator + ---');
  // 3. Implementasi operator +
  final initialDate = DateTime(2024, 1, 15);
  final durationToAdd = Duration(days: 10, hours: 3);
  final futureDate = initialDate + durationToAdd;

  print('Initial date: ${initialDate.toFormattedString()}');
  print('Duration to add: ${durationToAdd.inDays} days, ${durationToAdd.inHours % 24} hours');
  print('Future date: ${futureDate.toFormattedString()}');
  print('Future date (full): $futureDate');

  final anotherDuration = Duration(days: 2);
  final evenFutureDate = futureDate + anotherDuration;
  print('Future date + 2 day: ${evenFutureDate.toFormattedString()}');
}

🍃 Tipe yang Didukung Extension

  • 📌 Built-in types: String, int, double, List, Map, etc.
  • 📌 Custom classes: Kelas buatan sendiri
  • 📌 Generic types: List<T>, Map<K, V>
  • 📌 Nullable types: String?, int?
extension IntListExtensions on List<int> {
    int get sum => fold(0, (total, item) => total + item);
    double get average => isEmpty ? 0 : sum / length;
}
void main(){
    var numbers = [1, 2, 3, 4, 5];
    print(numbers.sum); // 15
    print(numbers.average); // 3.0
}

🛡️ Null Safety Extension (Rainforest Resilience)

extension NullableStringExtensions on String? {
    String orEmpty() => this ?? '';
    bool get isNullOrEmpty => this == null || this!.isEmpty;
    bool get isNotNullOrEmpty => !isNullOrEmpty;
    String withDefault(String defaultValue) => this ?? defaultValue;
}

void main(){
    String? name;
    print(name.isNullOrEmpty); // true
    print(name.withDefault('Guest')); // 'Guest'
}

💰 Praktik 1 - Utilities untuk Dart Types

extension NumberExtensions on num {
    String toRupiah() {
        return 'Rp${toStringAsFixed(0).replaceAllMapped(
            RegExp(r'(\d{1,3})(?=(\d{3})+(?!\d))'),
            (match) => '${match[1]}.',
        )}';
    }
    bool isBetween(num min, num max) => this >= min && this <= max;
    num power(int exponent) {
        var result = this;
        for (var i = 1; i < exponent; i++) result *= this;
        return result;
    }
}
void main(){
    var price = 1500000;
    print(price.toRupiah()); // Rp1.500.000
    print(5.isBetween(1, 10)); // true
    print(2.power(3)); // 8
}

🔁 Generic Extensions pada List<T>

extension ListExtensions<T> on List<T> {
    T? getOrNull(int index) => (index < 0 || index >= length) ? null : this[index];
    List<List<T>> chunk(int size) {
        var chunks = <List<T>>[];
        for (var i = 0; i < length; i += size) {
            chunks.add(sublist(i, i + size > length ? length : i + size));
        }
        return chunks;
    }
    List<T> distinct() {
        var result = <T>[];
        for (var item in this) if (!result.contains(item)) result.add(item);
        return result;
    }
}
void main(){
    var numbers = [1, 2, 3, 4, 5, 6, 7];
    print(numbers.chunk(3)); // [[1, 2, 3], [4, 5, 6], [7]]
    print([1, 2, 2, 3, 3, 3].distinct()); // [1, 2, 3]
}

// File: date_utils.dart
extension on DateTime {
    String get toYYYYMMDD => '$year-${month.toString().padLeft(2, '0')}-${day.toString().padLeft(2, '0')}';
    DateTime get startOfDay => DateTime(year, month, day);
}
void main() {
    var today = DateTime.now();
    print(today.toYYYYMMDD); // '2024-01-22'
    print(today.startOfDay);
}

🏝️ Ringkasan & Best Practices (Guardian Wisdom)

🌿 Ringkasan:

  • Extension methods: tambah method ke tipe yang ada
  • Sintaks: extension Nama on Type
  • Support: method, getter, setter, operator
  • Generic extensions: on List<T>

✅ Best Practices (Hutan Kode):

  • 📛 Nama jelas: StringExtensions, DateTimeUtils
  • 📁 Group by type: Satu file per tipe utama
  • 🚫 Avoid overuse: Jangan jadikan semua utility sebagai extension
  • 🛡️ Null safety: Handle nullable types dengan baik
  • 📝 Dokumentasi: Tambahkan komentar untuk public extensions
🌱 "Jaga kode tetap alami, berdayakan ekstensi tanpa merusak aslinya — seperti hutan yang tumbuh harmonis."
